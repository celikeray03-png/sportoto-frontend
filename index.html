async function uploadCoupon() { const pwd = document.getElementById('adminPassword').value; const title = document.getElementById('couponTitle').value.trim() || 'Spor Toto Kuponu'; const text = document.getElementById('couponText').value.trim(); const btn = document.querySelector("#adminModal button[onclick='uploadCoupon()']"); if (!pwd || !text) { alert("Lütfen şifre ve kupon metnini girin."); return; } btn.innerText = "Yayınlanıyor..."; btn.disabled = true; let parsedMatches = Array.from({ length: TOTAL_MATCHES }, (_, i) => ({ id: i + 1, home: `Maç ${i + 1}`, away: "Bekleniyor", status: "PENDING", homeScore: null, awayScore: null })); let parsedColumns = [Array(TOTAL_MATCHES).fill([]), Array(TOTAL_MATCHES).fill([]), Array(TOTAL_MATCHES).fill([])]; const lines = text.split('\n').filter(l => l.trim().length > 0); lines.forEach((line, idx) => { if (idx < TOTAL_MATCHES) { let cleanLine = line.replace(/^[0-9]+[\.\-\s]+/, ''); let parts = cleanLine.split('|'); let teamsPart = parts[0]; let teams = teamsPart.split(/\s+[-–vs]+\s+/i); if (teams.length < 2) teams = teamsPart.split(/[-–vs]/i); if (teams.length >= 2) { parsedMatches[idx].home = cleanTeamName(teams[0]); parsedMatches[idx].away = cleanTeamName(teams[1]); } if (parts.length > 1) { for (let c = 1; c < parts.length && c <= 3; c++) { let colText = parts[c]; let pickMatch = colText.match(/Kolon\s*\d+:\s*([0-12,\s]+)/i); if (pickMatch) { let picks = pickMatch[1].split(',').map(p => p.trim()).filter(p => ['1','0','2'].includes(p)); parsedColumns[c-1][idx] = picks; } } } } }); try { const res = await fetch(`${BACKEND_URL}/api/admin/update-coupon`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ password: pwd, title: title, matches_data: { matches: parsedMatches, columns: parsedColumns } }) }); const data = await res.json(); if (data.status === 'success') { alert("Kupon başarıyla güncellendi!"); closeAdminModal(); loadData(); } else { alert(data.message || "Hatalı şifre!"); } } catch (e) { alert("Sunucu uyanıyor olabilir, lütfen 10 saniye sonra tekrar deneyin."); } finally { btn.innerText = "Yayınla"; btn.disabled = false; } }
